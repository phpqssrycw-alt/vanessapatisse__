<script>
  // WhatsApp
  const WHATSAPP_NUMBER = "33603696752";

  // ✅ Bloque Jour J et lendemain = min = aujourd’hui + 2 jours
  const MIN_LEAD_DAYS = 2;

  // Parts layer cake
  const layerPartsBySize = {
    "15": "≈ 6–8 parts",
    "20": "≈ 10–12 parts",
    "30": "≈ 18–24 parts"
  };

  // Layer cake flavors
  const layerFlavorsNature = [
    "Framboise",
    "Passion",
    "Coco",
    "Chocolat",
    "Chocolat praliné",
    "Chantilly nature",
    "Chantilly & morceaux d’abricots",
    "Chantilly & morceaux de cocktail de fruits"
  ];

  // Base chocolat: 1 saveur OU 2 saveurs (chantilly + chocolat / chantilly + chocolat praliné)
  const layerFlavorsChoco = [
    "Chocolat",
    "Chocolat praliné",
    "Chantilly nature",
    "Chantilly + chocolat",
    "Chantilly + chocolat praliné"
  ];

  // Créoles pricing
  const CREOLE_BASE_PRICE = 35;
  const CREOLE_TWO_FLAVORS_EXTRA = 3;

  // Elements
  const elCategory = document.getElementById("category");
  const elQty = document.getElementById("qty");
  const elPickupDate = document.getElementById("pickupDate");
  const elPickupSlot = document.getElementById("pickupSlot");
  const elLeadAlert = document.getElementById("leadAlert");
  const elBtn = document.getElementById("btn");

  // Blocks
  const layerBlock = document.getElementById("layerBlock");
  const bucheBlock = document.getElementById("bucheBlock");
  const creoleBlock = document.getElementById("creoleBlock");

  // Layer
  const elLayerBase = document.getElementById("layerBase");
  const elLayerFlavor = document.getElementById("layerFlavor");
  const elLayerSize = document.getElementById("layerSize");
  const elLayerParts = document.getElementById("layerParts");

  // Buche
  const elBucheFlavor = document.getElementById("bucheFlavor");

  // Creole
  const elCreoleType = document.getElementById("creoleType");
  const elGaletteOptions = document.getElementById("galetteOptions");
  const elCreoleFlavor1 = document.getElementById("creoleFlavor1");
  const elTwoFlavors = document.getElementById("twoFlavors");
  const elFlavor2Block = document.getElementById("flavor2Block");
  const elCreoleFlavor2 = document.getElementById("creoleFlavor2");
  const elCreolePriceLine = document.getElementById("creolePriceLine");

  // Customer
  const elName = document.getElementById("name");
  const elNotes = document.getElementById("notes");

  function toISODate(d){
    const yyyy = d.getFullYear();
    const mm = String(d.getMonth()+1).padStart(2,"0");
    const dd = String(d.getDate()).padStart(2,"0");
    return `${yyyy}-${mm}-${dd}`;
  }

  function setMinPickupDate(){
    const now = new Date();
    const minDt = new Date(now.getFullYear(), now.getMonth(), now.getDate() + MIN_LEAD_DAYS); // +2 jours
    const minDateStr = toISODate(minDt);
    elPickupDate.min = minDateStr;

    if(!elPickupDate.value){
      elPickupDate.value = minDateStr;
    } else {
      // si une ancienne valeur < min, on corrige
      if(elPickupDate.value < minDateStr){
        elPickupDate.value = minDateStr;
      }
    }
  }

  function leadTimeOk(){
    if(!elPickupDate.value) return false;
    const now = new Date();
    const minDt = new Date(now.getFullYear(), now.getMonth(), now.getDate() + MIN_LEAD_DAYS);
    const minStr = toISODate(minDt);
    return elPickupDate.value >= minStr; // bloque jour J + lendemain
  }

  function showBlocks(){
    const cat = elCategory.value;
    layerBlock.style.display  = (cat === "Layer Cake") ? "block" : "none";
    bucheBlock.style.display  = (cat === "Gâteau roulé (bûche)") ? "block" : "none";
    creoleBlock.style.display = (cat === "Gâteaux créoles") ? "block" : "none";
  }

  function setLayerFlavorOptions(base){
    elLayerFlavor.innerHTML = "";
    const list = base === "Nature" ? layerFlavorsNature : (base === "Chocolat" ? layerFlavorsChoco : []);
    if(!list.length){
      elLayerFlavor.disabled = true;
      const opt = document.createElement("option");
      opt.textContent = "Choisir d’abord une base…";
      opt.disabled = true; opt.selected = true; opt.value = "";
      elLayerFlavor.appendChild(opt);
      return;
    }
    elLayerFlavor.disabled = false;

    const head = document.createElement("option");
    head.textContent = "Choisir…";
    head.disabled = true; head.selected = true; head.value = "";
    elLayerFlavor.appendChild(head);

    list.forEach(v=>{
      const opt = document.createElement("option");
      opt.value = v; opt.textContent = v;
      elLayerFlavor.appendChild(opt);
    });
  }

  function updateLayerParts(){
    const s = elLayerSize.value;
    if(layerPartsBySize[s]){
      elLayerParts.style.display = "inline-block";
      elLayerParts.textContent = `Nombre de parts : ${layerPartsBySize[s]}`;
    } else {
      elLayerParts.style.display = "none";
      elLayerParts.textContent = "";
    }
  }

  function updateCreoleUI(){
    const type = elCreoleType.value;

    if(type === "Gâteau à la nana"){
      elGaletteOptions.style.display = "none";
      elTwoFlavors.checked = false;
      elFlavor2Block.style.display = "none";
      elCreoleFlavor1.value = "";
      elCreoleFlavor2.value = "";
      elCreolePriceLine.innerHTML = `<b>Prix :</b> <b>${CREOLE_BASE_PRICE} €</b>`;
      return;
    }

    if(type === "Galette créole"){
      elGaletteOptions.style.display = "block";
      updateCreolePrice();
      return;
    }

    elGaletteOptions.style.display = "none";
    elCreolePriceLine.innerHTML = `<b>Prix :</b> —`;
  }

  function updateCreolePrice(){
    if(elCreoleType.value !== "Galette créole") return;
    const price = CREOLE_BASE_PRICE + (elTwoFlavors.checked ? CREOLE_TWO_FLAVORS_EXTRA : 0);
    elCreolePriceLine.innerHTML = `<b>Prix :</b> <b>${price} €</b>`;
  }

  function validateAndToggle(){
    // Alerte date
    if(elPickupDate.value && !leadTimeOk()){
      elLeadAlert.style.display = "block";
      elLeadAlert.textContent = "Commande impossible : Jour J et lendemain sont bloqués. Merci de choisir une date plus loin (minimum +48h).";
    } else {
      elLeadAlert.style.display = "none";
      elLeadAlert.textContent = "";
    }

    const qtyOk  = (parseInt(elQty.value, 10) >= 1);
    const catOk  = !!elCategory.value;
    const nameOk = !!elName.value.trim();
    const dateOk = !!elPickupDate.value && leadTimeOk();
    const slotOk = !!elPickupSlot.value;

    let productOk = false;

    if(elCategory.value === "Layer Cake"){
      productOk = !!(elLayerBase.value && elLayerFlavor.value && elLayerSize.value);
    } else if(elCategory.value === "Gâteau roulé (bûche)"){
      productOk = !!elBucheFlavor.value;
    } else if(elCategory.value === "Gâteaux créoles"){
      const type = elCreoleType.value;
      if(type === "Gâteau à la nana"){
        productOk = true;
      } else if(type === "Galette créole"){
        if(!elCreoleFlavor1.value) productOk = false;
        else if(elTwoFlavors.checked && !elCreoleFlavor2.value) productOk = false;
        else productOk = true;
      } else {
        productOk = false;
      }
    }

    elBtn.disabled = !(qtyOk && catOk && productOk && nameOk && dateOk && slotOk);
  }

  // Events
  elCategory.addEventListener("change", ()=>{ showBlocks(); validateAndToggle(); });
  elQty.addEventListener("input", validateAndToggle);

  // Layer (ordre : base -> goût -> taille)
  elLayerBase.addEventListener("change", ()=>{
    setLayerFlavorOptions(elLayerBase.value);
    // reset goût + taille quand on change de base (pour garder l’ordre)
    elLayerSize.value = "";
    updateLayerParts();
    validateAndToggle();
  });
  elLayerFlavor.addEventListener("change", validateAndToggle);
  elLayerSize.addEventListener("change", ()=>{ updateLayerParts(); validateAndToggle(); });

  // Buche
  elBucheFlavor.addEventListener("change", validateAndToggle);

  // Creole
  elCreoleType.addEventListener("change", ()=>{ updateCreoleUI(); validateAndToggle(); });
  elCreoleFlavor1.addEventListener("change", validateAndToggle);
  elTwoFlavors.addEventListener("change", ()=>{
    elFlavor2Block.style.display = elTwoFlavors.checked ? "block" : "none";
    if(!elTwoFlavors.checked) elCreoleFlavor2.value = "";
    updateCreolePrice();
    validateAndToggle();
  });
  elCreoleFlavor2.addEventListener("change", ()=>{ updateCreolePrice(); validateAndToggle(); });

  // Nom / Date / Slot
  elName.addEventListener("input", validateAndToggle);
  elPickupDate.addEventListener("change", validateAndToggle);
  elPickupSlot.addEventListener("change", validateAndToggle);

  // Send to WhatsApp
  elBtn.addEventListener("click", ()=>{
    const cat = elCategory.value;
    const qty = parseInt(elQty.value,10) || 1;

    const lines = [
      "Bonjour 😊, je souhaite passer commande :",
      `• Catégorie : ${cat}`,
      `• Quantité : ${qty}`,
      `• Nom : ${elName.value.trim()}`,
      `• Retrait : ${elPickupDate.value} — ${elPickupSlot.value}`
    ];

    if(cat === "Layer Cake"){
      const parts = layerPartsBySize[elLayerSize.value] || "—";
      lines.push(`• Produit : Layer Cake`);
      lines.push(`• Base : ${elLayerBase.value}`);
      lines.push(`• Goût : ${elLayerFlavor.value}`);
      lines.push(`• Taille : ${elLayerSize.value} cm (${parts})`);
      lines.push(`• Prix : à partir de 45 €`);
    }

    if(cat === "Gâteau roulé (bûche)"){
      lines.push(`• Produit : Gâteau roulé (bûche)`);
      lines.push(`• Longueur : ≈ 30 cm`);
      lines.push(`• Parts : ≈ 8–10 parts`);
      lines.push(`• Parfum : ${elBucheFlavor.value}`);
      lines.push(`• Prix : 25 €`);
    }

    if(cat === "Gâteaux créoles"){
      lines.push(`• Taille : ≈ 34 cm`);
      lines.push(`• Parts : ≈ 12–15 parts`);

      const type = elCreoleType.value;

      if(type === "Gâteau à la nana"){
        lines.push(`• Produit : Gâteau à la nana`);
        lines.push(`• Prix : ${CREOLE_BASE_PRICE} €`);
      } else {
        const two = elTwoFlavors.checked;
        const price = CREOLE_BASE_PRICE + (two ? CREOLE_TWO_FLAVORS_EXTRA : 0);
        lines.push(`• Produit : Galette créole`);
        if(two){
          lines.push(`• Parfums : ${elCreoleFlavor1.value} + ${elCreoleFlavor2.value}`);
        } else {
          lines.push(`• Parfum : ${elCreoleFlavor1.value}`);
        }
        lines.push(`• Prix : ${price} €`);
      }
    }

    const notes = elNotes.value.trim();
    if(notes) lines.push(`• Précisions : ${notes}`);

    lines.push("");
    lines.push("Merci ✨");

    const text = encodeURIComponent(lines.join("\n"));
    window.location.href = `https://wa.me/${WHATSAPP_NUMBER}?text=${text}`;
  });

  // Init
  setMinPickupDate();
  showBlocks();
  setLayerFlavorOptions("");
  updateLayerParts();
  updateCreoleUI();
  validateAndToggle();
</script>
</body>
