<script>
const wa = "33603696752";

const category = document.getElementById("category");
const send = document.getElementById("send");
const pickup = document.getElementById("pickup");
const pickupDate = document.getElementById("pickupDate");
const dateHint = document.getElementById("dateHint");

const layerFlavorsNature = [
  "Framboise","Passion","Coco","Chocolat","Chocolat praliné",
  "Chantilly nature","Chantilly & morceaux d’abricots","Chantilly & morceaux de cocktail de fruits"
];

const layerFlavorsChoco = [
  "Chocolat","Chocolat praliné","Chantilly nature",
  "Chantilly + chocolat","Chantilly + chocolat praliné"
];

// ✅ Règle 48h (silencieuse)
function leadTimeOk(){
  if(!pickupDate.value) return false;
  const selected = new Date(pickupDate.value + "T00:00:00");
  const min = new Date(Date.now() + 48*60*60*1000);
  const minDay = new Date(min.getFullYear(), min.getMonth(), min.getDate());
  return selected >= minDay;
}

// date min visible dans le calendrier
const min = new Date(Date.now() + 48*60*60*1000);
pickupDate.min = min.toISOString().split("T")[0];

function showSection(){
  document.getElementById("layer").style.display = category.value==="Layer Cake" ? "block" : "none";
  document.getElementById("buche").style.display = category.value==="Gâteau roulé (bûche)" ? "block" : "none";
  document.getElementById("creole").style.display = category.value==="Gâteaux créoles" ? "block" : "none";
}

category.onchange = ()=>{
  showSection();
  checkReady();
};

document.getElementById("layerBase").onchange = (e)=>{
  const list = e.target.value==="Nature" ? layerFlavorsNature : layerFlavorsChoco;
  document.getElementById("layerFlavor").innerHTML =
    `<option value="">Choisir…</option>` + list.map(v=>`<option>${v}</option>`).join("");
  checkReady();
};

document.getElementById("creoleType").onchange = (e)=>{
  const isGalette = e.target.value==="Galette créole";
  document.getElementById("galetteOptions").style.display = isGalette ? "block" : "none";
  // reset parfums si gâteau à l’ananas
  if(!isGalette){
    document.getElementById("twoFlavors").checked = false;
    document.getElementById("creoleFlavor1").value = "";
    document.getElementById("creoleFlavor2").value = "";
    document.getElementById("creoleFlavor2").style.display = "none";
  }
  checkReady();
};

document.getElementById("twoFlavors").onchange = (e)=>{
  document.getElementById("creoleFlavor2").style.display = e.target.checked ? "block" : "none";
  if(!e.target.checked) document.getElementById("creoleFlavor2").value = "";
  checkReady();
};

function checkReady(){
  // hint date discret
  if(pickupDate.value && !leadTimeOk()){
    dateHint.textContent = "Merci de choisir une date au moins 48h à l’avance.";
  } else {
    dateHint.textContent = "";
  }

  const baseOk = category.value && pickup.value && pickupDate.value && leadTimeOk();
  let productOk = true;

  if(category.value==="Layer Cake"){
    productOk = !!(
      document.getElementById("layerSize").value &&
      document.getElementById("layerBase").value &&
      document.getElementById("layerFlavor").value
    );
  }

  if(category.value==="Gâteau roulé (bûche)"){
    productOk = !!document.getElementById("bucheFlavor").value;
  }

  if(category.value==="Gâteaux créoles"){
    const type = document.getElementById("creoleType").value;
    if(!type) productOk = false;
    else if(type==="Gâteau à l’ananas"){
      productOk = true; // pas de parfum
    } else {
      // galette
      const f1 = document.getElementById("creoleFlavor1").value;
      const two = document.getElementById("twoFlavors").checked;
      const f2 = document.getElementById("creoleFlavor2").value;
      productOk = !!f1 && (!two || !!f2);
    }
  }

  send.disabled = !(baseOk && productOk);
}

["pickup","pickupDate","qty","layerSize","layerBase","layerFlavor","bucheFlavor","creoleType","creoleFlavor1","creoleFlavor2"].forEach(id=>{
  const el = document.getElementById(id);
  if(el) el.onchange = checkReady;
});

send.onclick = ()=>{
  if(send.disabled) return;

  const qty = document.getElementById("qty").value;
  let msg = `Bonjour 😊, je souhaite commander :\n`;
  msg += `Catégorie : ${category.value}\n`;
  msg += `Quantité : ${qty}\n`;
  msg += `Date de retrait : ${pickupDate.value}\n`;
  msg += `Créneau : ${pickup.value}\n`;

  if(category.value==="Layer Cake"){
    msg += `Taille : ${document.getElementById("layerSize").value}\n`;
    msg += `Base : ${document.getElementById("layerBase").value}\n`;
    msg += `Goût : ${document.getElementById("layerFlavor").value}\n`;
    msg += `Prix : à partir de 45 €\n`;
  }

  if(category.value==="Gâteau roulé (bûche)"){
    msg += `Parfum : ${document.getElementById("bucheFlavor").value}\n`;
    msg += `Prix : 25 €\n`;
  }

  if(category.value==="Gâteaux créoles"){
    const type = document.getElementById("creoleType").value;
    msg += `Produit : ${type}\n`;

    if(type==="Galette créole"){
      const two = document.getElementById("twoFlavors").checked;
      const f1 = document.getElementById("creoleFlavor1").value;
      const f2 = document.getElementById("creoleFlavor2").value;
      msg += two ? `Parfums : ${f1} + ${f2}\nPrix : 38 €\n` : `Parfum : ${f1}\nPrix : 35 €\n`;
    } else {
      msg += `Prix : 35 €\n`;
    }
  }

  const name = document.getElementById("name").value.trim();
  const notes = document.getElementById("notes").value.trim();
  if(name) msg += `Nom : ${name}\n`;
  if(notes) msg += `Précisions : ${notes}\n`;

  window.location.href = `https://wa.me/${wa}?text=` + encodeURIComponent(msg);
};

// init
showSection();
checkReady();
</script>
